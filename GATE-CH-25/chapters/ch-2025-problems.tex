\frac{dy}{dx}

\end{align}

Separating the variables,

\begin{align}
\frac{1}{y}\,dy
&=-\frac{1}{x}\,dx.
\end{align}

Integrating both sides,

\begin{align}
\int\frac{1}{y}\,dy
&=-\int\frac{1}{x}\,dx.
\end{align}

\begin{align}
\ln|y|
&=-\ln|x|+C.
\end{align}

\begin{align}
\ln|xy|
&=C,
\end{align}

\begin{align}
y&=\frac{C}{x}.
\end{align}

\begin{align}
y&=0.
\end{align}

Thus, both

\[
y=\frac{C}{x}
\qquad\text{and}\qquad
y=0
\]

are valid solutions.

\begin{align}
\frac{dy}{dx}
&=f(x,y)=-\frac{y}{x}.
\end{align}

For a step size $h$, the trapezoidal approximation is

\begin{align}
y_{n+1}
&=y_n+\frac{h}{2}
\left[
f(x_n,y_n)+f(x_{n+1},y_{n+1})
\right].
\end{align}

Substituting

\begin{align}
f(x,y)
&=-\frac{y}{x},
\end{align}

gives

\begin{align}
y_{n+1}
&=y_n+\frac{h}{2}
\left[
-\frac{y_n}{x_n}
-\frac{y_{n+1}}{x_{n+1}}
\right].
\end{align}

This relation is used iteratively to obtain the numerical solution,

which can then be compared with the theoretical solution
$y=C/x$.

The trapezoidal method can then be used to generate numerical points for plotting the solution family.

e^{-(6-1)/5}
\\
&=
\frac{2}{5e}.
\label{eq:q60_max_slope}
\end{align}

The maximum-slope method approximates the actual curve by the tangent
at the inflection point. Therefore,

\begin{align}
y-y_i=R(t-t_i).
\label{eq:q60_tangent}
\end{align}

Substituting \eqref{eq:q60_inflection_output},
\eqref{eq:q60_max_slope}, and \eqref{eq:q60_inflection} into
\eqref{eq:q60_tangent},

\begin{align}
y-\left(2-\frac{4}{e}\right)
=
\frac{2}{5e}(t-6).
\label{eq:q60_tangent_expanded}
\end{align}

The fitted dead time $\theta_m$ is the time-axis intercept of this
tangent. Therefore, putting $y=0$ in
\eqref{eq:q60_tangent_expanded},

\begin{align}
-\left(2-\frac{4}{e}\right)
&=
\frac{2}{5e}(\theta_m-6).
\end{align}

Hence,

\begin{align}
\theta_m
&=
6-\frac{2(1-2/e)}{2/(5e)}
\\
&=
6-5e+10
\\
&=
16-5e
\\
&\approx2.4086~\text{min}.
\label{eq:q60_dead_time}
\end{align}

For the fitted FOPDT model,

\begin{align}
G_m(s)
=
\frac{K e^{-\theta_m s}}
{\tau_m s+1},
\end{align}

the maximum slope is

\begin{align}
R=\frac{K}{\tau_m}.
\label{eq:q60_fopdt_slope}
\end{align}

Therefore,

\begin{align}
\tau_m
&=
\frac{K}{R}
\\
&=
\frac{2}{2/(5e)}
\\
&=
5e
\\
&\approx13.5914~\text{min}.
\label{eq:q60_time_constant}
\end{align}

Finally, using \eqref{eq:q60_dead_time} and
\eqref{eq:q60_time_constant},

\begin{align}
\frac{\tau_m}{\theta_m}
&=
\frac{5e}{16-5e}
\\
&=
\frac{13.5914}{2.4086}
\\
&\approx5.642.
\end{align}

\item The block diagram of a series cascade control system (with time in minutes) is shown in the figure. For $\tau_I=8$ min and $K_c^s=1$, the maximum value of $K_c^m$, below which the cascade control system is stable, is \underline{\hspace{2cm}} (rounded off to the nearest integer). \hfill (GATE~CH~2025)

\begin{figure}[H]
    \centering
    \includegraphics[width=1\linewidth]{figs/q62/Q62-Problem.pdf}
    \caption{Series cascade control system given in GATE CH 2025 Q62}
    \label{fig:ch25_q62_problem}
\end{figure}

\solution The inner process is
\begin{align}
G_1(s)=\frac{1}{2s+1},
\end{align}
and the inner controller gain is $K_c^s=1$. Therefore, the closed-loop inner transfer function is
\begin{align}
G_{\mathrm{inner}}(s)
&=\frac{K_c^sG_1(s)}{1+K_c^sG_1(s)}\\
&=\frac{1/(2s+1)}{1+1/(2s+1)}\\
&=\frac{1}{2(s+1)}.
\end{align}
The outer process is
\begin{align}
G_2(s)=\frac{2}{(8s+1)(4s+1)}.
\end{align}
Hence the equivalent process for the outer loop is
\begin{align}
G_{\mathrm{eq}}(s)
&=G_{\mathrm{inner}}(s)G_2(s)\\
&=\frac{1}{(s+1)(8s+1)(4s+1)}.
\end{align}
The outer controller is
\begin{align}
G_c(s)=K_c^m\frac{\tau_Is+1}{\tau_Is}.
\end{align}
Since $\tau_I=8$ min,
\begin{align}
G_c(s)=K_c^m\frac{8s+1}{8s}.
\end{align}
Therefore, the open-loop transfer function of the outer loop is
\begin{align}
G_c(s)G_{\mathrm{eq}}(s)
&=K_c^m\frac{8s+1}{8s(s+1)(8s+1)(4s+1)}\\
&=\frac{K_c^m}{8s(s+1)(4s+1)}.
\end{align}
For unity negative feedback, using \eqref{eq:unity_closed_loop}, the characteristic equation is
\begin{align}
1+\frac{K_c^m}{8s(s+1)(4s+1)}=0,
\end{align}
which gives
\begin{align}
8s(s+1)(4s+1)+K_c^m&=0\\
32s^3+40s^2+8s+K_c^m&=0.
\end{align}
Using the Routh-Hurwitz relations in \eqref{eq:routh_array}, the Routh array is
\begin{align}
\myvec{
32 & 8\\
40 & K_c^m\\
\dfrac{40(8)-32K_c^m}{40} & 0\\
K_c^m &
}.
\end{align}
For stability, every element in the first column must be positive. Thus,
\begin{align}
8-0.8K_c^m&>0,\\
K_c^m&>0.
\end{align}
Therefore,
\begin{align}
0<K_c^m<10.
\end{align}
Hence the maximum value of $k_c^m$ below which the system is stable, rounded to the nearest integer, is 10.
